cmake_minimum_required(VERSION 3.24)
project(QITestVendorAdapterTemplate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core)
if(CMAKE_CROSSCOMPILING AND QT_VERSION_MAJOR EQUAL 5)
    if(NOT EXISTS "${QITEST_QT5_HOST_ROOT}/bin/moc")
        message(FATAL_ERROR "Set QITEST_QT5_HOST_ROOT to the matching macOS Qt 5.12 host tools")
    endif()
    set_target_properties(Qt5::moc PROPERTIES IMPORTED_LOCATION "${QITEST_QT5_HOST_ROOT}/bin/moc")
endif()
add_library(QITestVendorAdapter MODULE VendorPlugin.cpp ../../src/device/IInstrumentAdapter.h)
target_include_directories(QITestVendorAdapter PRIVATE ../../src)
target_link_libraries(QITestVendorAdapter PRIVATE Qt${QT_VERSION_MAJOR}::Core)
# Cross-builds must set host moc exactly as in the parent project's toolchain.
